\begin{algorithm}[H]
\caption{$(\Obj{RolloutStorage}_g)_{g\in\mathcal G}\gets
\Obj{PartitionedQueueRollout}(\theta_k,\mathsf m,G_k)$}
\label{alg:partitioned-queue-rollout}
\DontPrintSemicolon
\KwParam{finite PBG
$(\mathcal S,\mathcal T,\mathcal P,o,(\mathcal E_\sigma)_\sigma,
\mathcal X,\mu,\chi)$; read-only $\pi^{\theta_k}$;
$\mathsf m\in\{\mathsf{normal},\mathsf{seqRB},\mathsf{branchRB}\}$;
GPU set $\mathcal G$; in sequential modes the common per-GPU root width $G_k$
chosen before current draws by \Obj{ChooseSequentialRootWidth} in
\texttt{sequential\_memory\_control.tex}}
\KwParam{finite opaque partition interface
$\mathfrak M=(\mathcal M,\kappa,\preceq,
(\Obj{PartitionSpec}_m)_{m\in\mathcal M})$:
$\kappa(\sigma)$ routes player nodes, $\preceq$ is an optional preorder, and
$\Obj{PartitionSpec}_m$ supplies natural input/model-row domains
$(\mathcal X_m,\mathcal Y_m,\varphi_m)$, a model boundary, fixed worst-case
bytes/work per row, and safe batch limits}
\KwParam{for $\mathsf{branchRB}$ only, public frozen
$\beta:\mathcal S\setminus\mathcal T^{\mathsf{pre-c}}\to[1,\infty)$}
\KwDef{$\mathcal T_{\mathsf{normal}}=\mathcal T$ and
$\mathcal T_{\mathsf{seqRB}}=\mathcal T_{\mathsf{branchRB}}
=\mathcal T^{\mathsf{pre-c}}$;
$n(s)=1$ in sequential modes and
\[
 n(s)=\lfloor\beta(\ell(s))\rfloor+
 \operatorname{Bernoulli}\!\left(\beta(\ell(s))-\lfloor\beta(\ell(s))\rfloor;
 K(k,\iota(s),\mathsf{round})\right)
\]
in $\mathsf{branchRB}$, including at the distinguished root $\varnothing$}
\KwDef{$\iota(s)$ is a placement-independent logical address.  Every random
draw uses a disjoint counter key $K(k,\iota(s),u,\mathsf{purpose})$; hence queue
order, physical pack size, and GPU speed cannot change the sampled object}
\KwDef{In sequential modes $\mathcal I_g=\{(g,i):0\leq i<G_k\}$.
\Obj{SafeAdmit} controls only how many of these already-fixed roots are live at
once; it may neither change $G_k$ nor discard, retry, or replace a trajectory}
\KwDef{$\Obj{Pack}_m(Q)$ removes a stable insertion-order prefix of $Q$.
$\Obj{PhysicalPlan}_m(B)$ returns stable maximum-feasible subspans of $B$;
it neither sorts nor reroutes rows.  $\Obj{Full}_m(Q)$ and all memory admission
tests use the partition's declared worst-case physical contract, not an
undeclared game label}
\KwDef{Any failed hard transaction sets a local failure flag and exits current
work without retrying a draw.  Every rank reaches the single common-order
failure agreement after local rollout; no rank enters targets or mass
collectives with a partial store}
\KwDef{$\operatorname{Min}_{\preceq}C=
\{m\in C:\nexists m'\in C,\ m'\prec m\}$.  Among full queues choose an element
of $\operatorname{Min}_{\preceq}C$ by decreasing $(\Obj{load},\Obj{age})$;
when forced to drain partial work, first restrict to the minimal nonempty
preorder layer and then maximize $(\Obj{fill},\Obj{age})$}
\KwOut{one GPU-local immutable $\Obj{RolloutStorage}_g$ containing only a
mode-specific compact topology, stable partition tables, and leaf rows}

\ForEach{$g\in\mathcal G$ \textbf{in parallel}}{
  Allocate a chunked device $\Obj{OccurrenceArena}_g$, append-only partition
  tables, bounded reusable staging, $f_g\gets0$, and
  $\Q{g}{m}=\varnothing$ for all $m$;
  \uIf{$\mathsf m=\mathsf{branchRB}$}{
    Create exactly one occurrence $s_0=\varnothing$ with
    $\ell(s_0)=\varnothing$, reach $r_{s_0,q}=1$, and
    $\log\omega(s_0)=0$; set $\iota(s_0)=(g,\varnothing)$; append
    $(P,L)\gets\Close_g(\{s_0\},\mathsf m)$ and enqueue each
    $s\in P$ into $\Q{g}{\kappa(\ell(s))}$;
    $U\gets\varnothing$;
  }
  \Else{
    $U\gets\mathcal I_g$
    \tcc*[r]{unstarted root trajectories}
  }

  \While{$U\ne\varnothing$ or $\bigsqcup_m\Q{g}{m}\ne\varnothing$}{
    $F\gets\{m:\Obj{Full}_m(\Q{g}{m})\}$;
    \uIf{$F\ne\varnothing$}{
      $m\gets\argmax_{m\in\operatorname{Min}_{\preceq}F}
      (\Obj{load}_m,\Obj{age}_m)$;
    }
    \uElseIf{$\mathsf m\ne\mathsf{branchRB}$ and
    $a=\max\{j\leq|U|:\Obj{SafeAdmit}_g(j)\}>0$}{
      Remove the next $a$ indices $i$ from $U$; in one device batch create
      roots $s_i$ with $\ell(s_i)=\varnothing$, reach $1$, sampled
      $X_i\sim\mu$ keyed by $(k,g,i,\mathsf{private})$, and
      $\iota(s_i)=(g,i,\varnothing)$;
      $(P,L)\gets\Close_g(\{s_i\}_i,\mathsf m)$; enqueue $P$ by $\kappa$;
      \textbf{continue};
    }
    \ElseIf{$\bigsqcup_m\Q{g}{m}\ne\varnothing$}{
      $E\gets\{m:\Q{g}{m}\ne\varnothing\}$;
      $m\gets\argmax_{m\in\operatorname{Min}_{\preceq}E}
      (\Obj{fill}_m,\Obj{age}_m)$;
    }
    \Else{
      Set $f_g\gets1$ and exit the loop: the next fixed root cannot be admitted
      under the declared transaction; do not replace, retry, or reroll it;
    }

    $B\gets\Obj{Pack}_m(\Q{g}{m})$; mark $B$ $\mathsf{EVALUATING}$;
    $R\gets\Reserve_g(B,\mathsf m)$
    \tcc*[r]{logical child/action addresses are fixed before physical packing}
    \ForEach{$J\in\Obj{PhysicalPlan}_m(B)$}{
      $Y_J\gets\Forward_m(\Obj{PartitionInputTable}_{g,m}[d(J)];\theta_k)$;
      $\Obj{DecisionTable}_{g,m}[d(J)].\mathtt{oldV}
        \gets
        \begin{cases}[l"l]
          Y_J.V[\varphi_m(\mathtt{realizedType})],
            &\mathsf m=\mathsf{normal},\\
          Y_J.V[1:Y_m],&\textnormal{otherwise};
        \end{cases}$;
      $F_J\gets\Materialize_g(J,Y_J,R,\mathsf m)$;
      $(P_J,L_J)\gets\Close_g(F_J,\mathsf m)$; enqueue $P_J$ by $\kappa$;
      $\Release(Y_J,\textnormal{sampler scratch for }J,
      \textnormal{physical input/output staging for }J)$;
    }
    Mark $B$ $\mathsf{DONE}$ and release each parent's mutable public-state and
    active reach/profile payload once its retained rows and all children are
    committed; retain only the occurrence's compact topology fields;
    $\operatorname{assert}(\textnormal{no }\mathsf{DONE}\textnormal{
      occurrence owns a mutable public state or active reach/profile})$;
    $\operatorname{assert}(\bigsqcup_m\Q{g}{m}
      =\{\textnormal{all live queued player occurrences}\})$;
    $\operatorname{assert}(\textnormal{no chance occurrence is pending})$;
  }

  \If{$f_g=0$}{
    $\Obj{RolloutStorage}_g\gets\Seal_g(\mathsf m)$; seal all stable $(m,d)$
    addresses and branched action spans; on seal transaction failure set
    $f_g\gets1$;
    $\operatorname{assert}(\textnormal{no occurrence owns a mutable public-state
      handle})$; compute partition-local fixed decision masses;
  }
  $\Release((\Q{g}{m})_m,\textnormal{active reaches/profiles},
  \textnormal{mutable public states},\textnormal{rollout staging},
  \Obj{OccurrenceArena}_g)$;
}
$f\gets\AllReduceMax_g f_g$ in the common control-collective order;
\If{$f=1$}{release every current partial/sealed store on every rank; atomically
checkpoint the last completed iteration and terminate cleanly}
\tcc{No model/queue collective occurs during rollout.  The boundary failure
agreement prevents a later collective deadlock.  No clone of $\theta_k$ is
needed: synchronous optimization cannot start until all fixed targets exist.}
\Return{$(\Obj{RolloutStorage}_g)_{g\in\mathcal G}$};
\end{algorithm}

\begin{algorithm}[H]
\caption{$\Obj{RolloutStorage}_g\gets\Obj{SealRolloutStorage}_g(\mathsf m)$}
\label{alg:seal-rollout-storage}
\DontPrintSemicolon
\KwIn{completed chunked occurrence, partition, action, and leaf arenas; exact
logical row counts and immutable schema; no queued or evaluating occurrence}
\KwOut{contiguous immutable topology and column blocks with no spare logical
rows; source chunks released only after their last copy}
Compute exact destination extents from realized counts.  Before allocating any
destination, reserve transactionally the simultaneous source-chunk capacity,
all destination bytes, and the declared bounded seal scratch under the hard
phase ceiling; on failure stop cleanly without changing the sampled object;
\uIf{$\mathsf m\ne\mathsf{branchRB}$}{
  Count player decisions per root in logical chronological order; exclusive-scan
  into $\mathtt{trajectoryPtr}[G_k+1]$ and scatter each retained $(m,d)$ using
  its immutable root id and within-trajectory decision ordinal; write the one
  aligned leaf row per root;
}
\Else{
  Copy the already stable parent/child-span columns into
  $\Obj{BranchedTreeSpan}$; validate that the unique root is row zero, every
  child span is contiguous, and every player action span has identical order;
}
For each partition and leaf column, allocate exactly the realized dense extent
required by its dtype/layout and copy chunk spans in stable row order using one
bounded reusable scratch span; do not use concatenation or allocator growth;
Synchronize seal copies, validate all pointers/extents and the byte ledger, then
release source chunks and seal scratch.  Retain destination blocks only;
\tcc{The hard sealing peak includes source capacity plus destination capacity
plus scratch.  Logical payload formulas alone are not an admission bound.}
\Return{the immutable sealed $\Obj{RolloutStorage}_g$};
\end{algorithm}

\begin{algorithm}[H]
\caption{$R\gets\Obj{ReserveChildren}_g(B,\mathsf m)$}
\label{alg:reserve-logical-children}
\DontPrintSemicolon
\KwIn{stable logical batch $B=(s_j)_{j=1}^b$ of player or chance parents}
\KwOut{reservation $R$ containing immutable child spans, logical RNG addresses,
and branched player-action spans}
$n_j\gets n(s_j)$ for $j=1,\ldots,b$; prefix-sum $(n_j)_j$;
Reserve exactly $\sum_jn_j$ child occurrences in stable $(j,u)$ order and write
$\mathtt{childBegin}[s_j]$, $\mathtt{degree}[s_j]=n_j$; fail rather than reduce
$n_j$ when the hard memory transaction cannot be admitted;
\uIf{$o(\ell(s_j))\in\mathcal P$ and $\mathsf m=\mathsf{branchRB}$}{
  For each $j$, reserve a contiguous span of $n_j$ rows in
  $\Obj{ActionTable}_{\kappa(\ell(s_j))}$, set its CSR pointer, and align its
  $u$th action row with the $u$th reserved child;
}
\ElseIf{$o(\ell(s_j))\in\mathcal P$}{
  Assert $n_j=1$ and address its implicit edge by the parent's stable decision row;
}
\ForEach{$(j,u):1\leq u\leq n_j$}{
  $\iota(s_{j,u})\gets(\iota(s_j),u)$;
  $\log\omega(s_{j,u})\gets\log\omega(s_j)-\log n_j$ in
  $\mathsf{branchRB}$;
}
\Return{$R$};
\end{algorithm}

\begin{algorithm}[H]
\caption{$F_J\gets\Obj{Materialize}_g(J,Y_J,R,\mathsf m)$}
\label{alg:materialize-reserved-children}
\DontPrintSemicolon
\KwIn{one physical parent subspan $J$, its current-policy output $Y_J$, and
pre-reserved logical child/action addresses}
\ForEach{reserved $(s_j,u)$ with $s_j\in J$ \textbf{in bounded device batches}}{
  $X_{j,u}\gets X_{i(j)}$ in sequential modes; otherwise sample
  $X_{j,u}\sim\mathbb P^{r_{s_j}}(\cdot\mid\ell(s_j))$ with its logical key;
  Sample $e_{j,u}$ from $Y_J$ at
  $\varphi_m(X_{j,u,o(\ell(s_j))})$ for a player parent,
  or from $\chi(\cdot\mid X_{j,u},\ell(s_j))$ for a chance parent;
  Set
  \[
    f_{j,u}(x)=
    \begin{cases}[l"l]
      Y_J.\pi[e_{j,u},\varphi_m(x)],&o(\ell(s_j))\in\mathcal P,\\
      \chi(e_{j,u}\mid x,\ell(s_j)),&o(\ell(s_j))=\mathsf c;
    \end{cases}
  \]
  Materialize the reserved child with public label $\ell(s_j)e_{j,u}$ and
  \[
    r_{s_{j,u},q}=r_{s_j,q}\ (q\ne o(\ell(s_j))),\qquad
    r_{s_{j,u},o(\ell(s_j))}=\operatorname{Normalize}
    \bigl(r_{s_j,o(\ell(s_j))}\odot
    f_{j,u}(-)\bigr).
  \]
  For a player parent, write $e_{j,u}$ and
  \[
    \mathtt{oldLogP}\gets
    \begin{cases}[l"l]
      \log Y_J.\pi[e_{j,u},\varphi_m(\mathtt{realizedType})],
        &\mathsf m=\mathsf{normal},\\
      \log Y_J.\pi[e_{j,u},1:Y_m],&\textnormal{otherwise}
    \end{cases}
  \]
  to its reserved action row (folded into its decision row in sequential modes);
}
Discard branched posterior profiles $X_{j,u}$ and every unselected old-policy
quantity not required by a registered loss;
\Return{the reserved children $F_J$ in logical order};
\end{algorithm}

\begin{algorithm}[H]
\caption{$(P,L)\gets\Obj{ChanceClose}_g(F,\mathsf m)$}
\label{alg:chance-close}
\DontPrintSemicolon
\KwIn{new occurrences $F$}
\KwOut{queued player occurrences $P$ and stopped leaf rows $L$}
$P,L\gets\varnothing$;
\While{$F\ne\varnothing$}{
  Split $F=F_T\sqcup F_P\sqcup F_C$ by
  $\ell(s)\in\mathcal T_{\mathsf m}$, player owner, or chance owner;
  \ForEach{$s\in F_P$}{
    $m\gets\kappa(\ell(s))$; append stable aligned rows $(m,d)$ to
    $\Obj{DecisionTable}_m$ and $\Obj{PartitionInputTable}_m$;
    write owner, normal-mode realized type, $\log\omega(s)$ when branched, and
    the natural $\Obj{Input}_m(\ell(s),r_s)$; store $(m,d)$ in $s$; append $s$ to $P$;
  }
  \ForEach{$s\in F_T$}{
    Append one leaf row.  In normal mode write the exact sampled terminal reward;
    in either RB mode write the exact generated reach, immutable completion
    input, and logical completion key.  Store the leaf address in $s$ and append
    it to $L$; release its mutable public-state and active reach/profile payload;
  }
  $F\gets\varnothing$;
  \ForEach{stable maximum-feasible $C\subseteq F_C$}{
    $R_C\gets\Reserve_g(C,\mathsf m)$;
    $F\gets F\sqcup\Materialize_g(C,\bot,R_C,\mathsf m)$;
    release every $s\in C$'s mutable public-state and active reach/profile
    payload after all its children are committed
    \tcc*[r]{chance needs no model output}
  }
}
\Return{$(P,L)$};
\end{algorithm}
